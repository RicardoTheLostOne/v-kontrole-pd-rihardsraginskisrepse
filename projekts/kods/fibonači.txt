#include <iostream>
#include <vector>

int main() {
    const int N = 100; 
    std::vector<unsigned long long> fib(N);

    // Pirmie divi Fibonači skaitļi
    fib[0] = 0;
    fib[1] = 1;

    // Aprēķinām virkni
    for (int i = 2; i < N; i++) {
        fib[i] = fib[i - 1] + fib[i - 2];
    }

    // Izvadām apgrieztā secībā
    for (int i = N - 1; i >= 0; i--) {
        std::cout << fib[i] << " ";
    }

    std::cout << std::endl;
    return 0;
}
