#include <iostream>
#include <string>

int main() {
    std::string teksts = "Kad skolēni sabojā git projektu: \"Nu ko… laiks jaunam klonam.\"";

    int skaits = 0;
    for (char c : teksts) {
        if (c == 'a' || c == 'A') { 
            skaits++;
        }
    }

    std::cout << "Burtu 'a' skaits: " << skaits << std::endl;

    return 0;
}
